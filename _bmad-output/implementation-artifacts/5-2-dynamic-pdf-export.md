# Story 5.2: Dynamic PDF Export

Status: ready-for-dev

## Story

As a user who has reviewed their report,
I want to download a PDF formatted to my state's column structure with correct page orientation, attestation, and disclaimer,
so that I have a document ready to print and submit.

## Acceptance Criteria

1. `GET /api/report/pdf` resolves stateConfig server-side and passes `report_columns` to `<ReportTablePDF>`
2. `src/app/api/report/pdf/report-table.pdf.tsx` is the **only** file in the project that imports `@react-pdf/renderer`
3. `<ReportTablePDF>` imports `formatCell` from `src/lib/report/format-cell.ts` — no duplicated formatting logic
4. Portrait orientation for ≤9 `report_columns`; landscape for ≥10
5. Column headers repeat at the top of each page
6. Attestation page: attestation language, signature lines, notarization block (if `log_notarization_required`)
7. Every page footer: "Generated by studentdriver.site. Not an official state document." plus Tier 2 line if applicable
8. Downloaded filename: `driving-log-[STATE_CODE]-[YYYY-MM-DD].pdf`
9. `npm run build` passes

### report-table.pdf.tsx Structure

```tsx
// src/app/api/report/pdf/report-table.pdf.tsx
import { Document, Page, View, Text, StyleSheet } from '@react-pdf/renderer';
import type { ReportColumn } from '@/lib/types/state-config';
import type { Trip } from '@/db/schema';
import { formatCell, type FormatCellContext } from '@/lib/report/format-cell';

const GREEN = '#1a5c2e';

const s = StyleSheet.create({
  page: { fontFamily: 'Helvetica', fontSize: 7, padding: 32 },
  headerRow: { flexDirection: 'row', backgroundColor: GREEN },
  row: { flexDirection: 'row' },
  rowAlt: { flexDirection: 'row', backgroundColor: '#e8f2eb' },
  cell: { borderRight: '0.5px solid #999', borderBottom: '0.5px solid #999', paddingHorizontal: 3, paddingVertical: 3 },
  cellHeader: { borderRight: '0.5px solid rgba(255,255,255,0.4)', paddingHorizontal: 3, paddingVertical: 4 },
  cellHeaderText: { color: '#fff', fontFamily: 'Helvetica-Bold', fontSize: 6, textAlign: 'center' },
  cellText: { color: '#111' },
  footer: { position: 'absolute', bottom: 20, left: 32, right: 32, fontSize: 6, color: '#888' },
});

type Props = {
  columns: ReportColumn[];
  trips: Trip[];
  stateName: string;
  officialFormName: string | null;
  attestationLanguage: string;
  logNotarizationRequired: boolean;
  isLandscape: boolean;
  tier: number;
  studentName: string | null;
  parentName: string | null;
  printedDate: string;
  stateCode: string;
};

export function ReportTablePDF({ columns, trips, ...props }: Props) {
  let cumulativeDay = 0;
  let cumulativeNight = 0;
  const rows = trips.map(trip => {
    cumulativeDay += trip.daytimeMinutes;
    cumulativeNight += trip.nighttimeMinutes;
    const ctx: FormatCellContext = { cumulativeDay, cumulativeNight, cumulativeTotal: cumulativeDay + cumulativeNight };
    return { trip, ctx };
  });

  const orientation = props.isLandscape ? 'landscape' : 'portrait';

  return (
    <Document>
      <Page size="LETTER" orientation={orientation} style={s.page}>
        {/* Header */}
        <View style={{ alignItems: 'center', marginBottom: 10 }}>
          <Text style={{ fontSize: 12, fontFamily: 'Helvetica-Bold' }}>
            {props.stateName} Practice Driving Log
          </Text>
          {props.officialFormName && (
            <Text style={{ fontSize: 7, color: '#555' }}>Reference: {props.officialFormName}</Text>
          )}
          <View style={{ flexDirection: 'row', gap: 24, fontSize: 7, marginTop: 4 }}>
            {props.studentName && <Text>Student: {props.studentName}</Text>}
            {props.parentName && <Text>Parent/Guardian: {props.parentName}</Text>}
            <Text>Printed: {props.printedDate}</Text>
          </View>
        </View>

        {/* Table */}
        <View>
          <View style={s.headerRow}>
            {columns.map(col => (
              <View key={col.label} style={[s.cellHeader, col.pdf_width_pt ? { width: col.pdf_width_pt } : { flex: 1 }]}>
                <Text style={s.cellHeaderText}>{col.label.toUpperCase()}</Text>
              </View>
            ))}
          </View>
          {rows.map(({ trip, ctx }, i) => (
            <View key={trip.id} style={i % 2 === 1 ? s.rowAlt : s.row}>
              {columns.map(col => (
                <View key={col.label} style={[s.cell, col.pdf_width_pt ? { width: col.pdf_width_pt } : { flex: 1 }]}>
                  <Text style={s.cellText}>{formatCell(trip, col, ctx)}</Text>
                </View>
              ))}
            </View>
          ))}
        </View>

        {/* Footer on every page */}
        <Text style={s.footer} fixed>
          Generated by studentdriver.site. Not an official state document.
          {props.tier === 2 ? ' Requirements based on publicly available information (IIHS). Verify with your state\'s licensing agency.' : ''}
        </Text>
      </Page>

      {/* Attestation page */}
      <Page size="LETTER" orientation={orientation} style={s.page}>
        <Text style={{ fontFamily: 'Helvetica-Bold', marginBottom: 16 }}>Attestation</Text>
        <Text style={{ marginBottom: 24, lineHeight: 1.5 }}>{props.attestationLanguage}</Text>
        <View style={{ flexDirection: 'row', gap: 40, marginTop: 24 }}>
          <View style={{ flex: 1 }}>
            <View style={{ borderBottom: '0.5px solid #333', marginBottom: 3 }}><Text> </Text></View>
            <Text style={{ fontSize: 6, color: '#555' }}>Student Signature &amp; Date</Text>
          </View>
          <View style={{ flex: 1 }}>
            <View style={{ borderBottom: '0.5px solid #333', marginBottom: 3 }}><Text> </Text></View>
            <Text style={{ fontSize: 6, color: '#555' }}>Parent / Guardian Signature &amp; Date</Text>
          </View>
        </View>
        {props.logNotarizationRequired && (
          <View style={{ marginTop: 32, borderTop: '0.5px solid #999', paddingTop: 16 }}>
            <Text style={{ fontFamily: 'Helvetica-Bold', marginBottom: 8 }}>Notary Public</Text>
            {/* Name, commission #, expiration, seal, date fields */}
          </View>
        )}
        <Text style={s.footer} fixed>
          Generated by studentdriver.site. Not an official state document.
        </Text>
      </Page>
    </Document>
  );
}
```

### route.tsx Changes

The current `route.tsx` (or `route.ts`) renders `<ReportDocument>` from `document.tsx`. Replace with `<ReportTablePDF>`:

```typescript
// src/app/api/report/pdf/route.tsx
import { renderToBuffer } from '@react-pdf/renderer';
import { STATE_CONFIGS } from '@/data/state-configs';
import { ReportTablePDF } from './report-table.pdf';

export async function GET() {
  // ... auth, fetch user/trips same as before ...
  const stateConfig = STATE_CONFIGS[user.stateCode ?? 'IL'];
  const isLandscape = stateConfig.report_columns.length >= 10;
  const today = new Date().toISOString().split('T')[0];

  const buffer = await renderToBuffer(
    <ReportTablePDF
      columns={stateConfig.report_columns}
      trips={tripRows}
      stateName={stateConfig.state_name}
      officialFormName={stateConfig.official_form_name}
      attestationLanguage={stateConfig.attestation_language}
      logNotarizationRequired={stateConfig.log_notarization_required}
      isLandscape={isLandscape}
      tier={stateConfig.tier}
      studentName={studentName}
      parentName={parentName}
      printedDate={today}
      stateCode={stateConfig.state_code}
    />
  );

  return new Response(buffer, {
    headers: {
      'Content-Type': 'application/pdf',
      'Content-Disposition': `attachment; filename="driving-log-${stateConfig.state_code}-${today}.pdf"`,
    },
  });
}
```
